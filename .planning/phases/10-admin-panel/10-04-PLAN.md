import { Card } from '@/components/ui/card';
import { Coffee, Image, CheckCircle, XCircle } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';
// Note: If date-fns is not installed, use relative date manually

interface ActivityItem {
  id: string;
  type: 'submission' | 'photo';
  action: 'approved' | 'declined' | 'rejected';
  itemName: string;
  timestamp: string;
}

interface RecentActivityProps {
  activities: ActivityItem[];
  translations: Record<string, string>;
}

export function RecentActivity({ activities, translations }: RecentActivityProps) {
  if (activities.length === 0) {
    return (
      <Card className="p-4">
        <p className="text-muted-foreground text-sm">
          {translations['admin.activity.empty']}
        </p>
      </Card>
    );
  }

  return (
    <Card className="p-4">
      <h3 className="font-semibold mb-3">{translations['admin.activity.title']}</h3>
      <ul className="space-y-3">
        {activities.map((activity) => (
          <li key={activity.id} className="flex items-center gap-3 text-sm">
            {/* Icon based on type */}
            {activity.type === 'submission' ? (
              <Coffee className="h-4 w-4 text-muted-foreground" />
            ) : (
              <Image className="h-4 w-4 text-muted-foreground" />
            )}

            {/* Action icon */}
            {activity.action === 'approved' ? (
              <CheckCircle className="h-4 w-4 text-green-500" />
            ) : (
              <XCircle className="h-4 w-4 text-red-500" />
            )}

            {/* Description */}
            <span className="flex-1">
              {activity.action === 'approved'
                ? translations['admin.activity.approved']
                : translations['admin.activity.rejected']
              }{' '}
              <span className="font-medium">{activity.itemName}</span>
            </span>

            {/* Timestamp */}
            <span className="text-xs text-muted-foreground">
              {formatRelativeTime(activity.timestamp)}
            </span>
          </li>
        ))}
      </ul>
    </Card>
  );
}

// Simple relative time function (avoids date-fns dependency)
function formatRelativeTime(timestamp: string): string {
  const now = new Date();
  const then = new Date(timestamp);
  const diffMs = now.getTime() - then.getTime();
  const diffMins = Math.floor(diffMs / 60000);
  const diffHours = Math.floor(diffMs / 3600000);
  const diffDays = Math.floor(diffMs / 86400000);

  if (diffMins < 1) return 'just now';
  if (diffMins < 60) return `${diffMins}m ago`;
  if (diffHours < 24) return `${diffHours}h ago`;
  return `${diffDays}d ago`;
}

import { createClient } from '@/lib/supabase/server';
import { AdminStats } from '@/components/admin/admin-stats';
import { RecentActivity } from '@/components/admin/recent-activity';
import { getTranslation } from '@/lib/i18n/translations';

export default async function AdminDashboard() {
  const supabase = await createClient();

  // Fetch submission counts by status
  const [
    { count: pendingSubmissions },
    { count: approvedSubmissions },
    { count: declinedSubmissions },
  ] = await Promise.all([
    supabase.from('cafe_submissions').select('*', { count: 'exact', head: true }).eq('status', 'pending'),
    supabase.from('cafe_submissions').select('*', { count: 'exact', head: true }).eq('status', 'approved'),
    supabase.from('cafe_submissions').select('*', { count: 'exact', head: true }).eq('status', 'declined'),
  ]);

  // Fetch photo counts by status
  const [
    { count: pendingPhotos },
    { count: approvedPhotos },
    { count: rejectedPhotos },
  ] = await Promise.all([
    supabase.from('photos').select('*', { count: 'exact', head: true }).eq('status', 'pending'),
    supabase.from('photos').select('*', { count: 'exact', head: true }).eq('status', 'approved'),
    supabase.from('photos').select('*', { count: 'exact', head: true }).eq('status', 'rejected'),
  ]);

  // Fetch recent activity (approved/declined submissions + approved/rejected photos)
  const [
    { data: recentSubmissions },
    { data: recentPhotos },
  ] = await Promise.all([
    supabase
      .from('cafe_submissions')
      .select('id, name, status, updated_at')
      .in('status', ['approved', 'declined'])
      .order('updated_at', { ascending: false })
      .limit(5),
    supabase
      .from('photos')
      .select('id, cafe:cafes(name), status, updated_at')
      .in('status', ['approved', 'rejected'])
      .order('updated_at', { ascending: false })
      .limit(5),
  ]);

  // Combine and sort activities
  const activities = [
    ...(recentSubmissions || []).map(s => ({
      id: `sub-${s.id}`,
      type: 'submission' as const,
      action: s.status as 'approved' | 'declined',
      itemName: s.name?.en || s.name?.ko || 'Unknown',
      timestamp: s.updated_at,
    })),
    ...(recentPhotos || []).map(p => ({
      id: `photo-${p.id}`,
      type: 'photo' as const,
      action: p.status === 'rejected' ? 'rejected' as const : 'approved' as const,
      itemName: p.cafe?.name?.en || p.cafe?.name?.ko || 'Unknown cafe',
      timestamp: p.updated_at,
    })),
  ]
    .sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime())
    .slice(0, 10);

  // Get translations
  const lang = 'en'; // TODO: get from context
  const translations = {
    'admin.title': getTranslation(lang, 'admin.title'),
    'admin.stats.submissions': getTranslation(lang, 'admin.stats.submissions'),
    'admin.stats.photos': getTranslation(lang, 'admin.stats.photos'),
    'admin.stats.pending': getTranslation(lang, 'admin.stats.pending'),
    'admin.stats.approved': getTranslation(lang, 'admin.stats.approved'),
    'admin.stats.declined': getTranslation(lang, 'admin.stats.declined'),
    'admin.stats.rejected': getTranslation(lang, 'admin.stats.rejected'),
    'admin.activity.title': getTranslation(lang, 'admin.activity.title'),
    'admin.activity.empty': getTranslation(lang, 'admin.activity.empty'),
    'admin.activity.approved': getTranslation(lang, 'admin.activity.approved'),
    'admin.activity.rejected': getTranslation(lang, 'admin.activity.rejected'),
  };

  return (
    <div className="space-y-8">
      <h1 className="text-2xl font-bold">{translations['admin.title']}</h1>

      {/* Stats Section */}
      <AdminStats
        submissions={{
          pending: pendingSubmissions || 0,
          approved: approvedSubmissions || 0,
          declined: declinedSubmissions || 0,
        }}
        photos={{
          pending: pendingPhotos || 0,
          approved: approvedPhotos || 0,
          rejected: rejectedPhotos || 0,
        }}
        translations={translations}
      />

      {/* Recent Activity Section */}
      <div>
        <RecentActivity activities={activities} translations={translations} />
      </div>
    </div>
  );
}
