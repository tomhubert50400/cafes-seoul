const approvePhotoSchema = z.object({
  photoId: z.string().uuid(),
});

export async function approvePhoto(input: z.infer<typeof approvePhotoSchema>): Promise<{
  success: boolean;
  error?: string;
}> {
  // 1. Verify authentication
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();

  if (!user) {
    return { success: false, error: 'Authentication required' };
  }

  // 2. Verify admin role
  if (!(await verifyAdminRole(supabase, user.id))) {
    return { success: false, error: 'Unauthorized - admin role required' };
  }

  // 3. Validate input
  const validation = approvePhotoSchema.safeParse(input);
  if (!validation.success) {
    return { success: false, error: validation.error.issues.map(i => i.message).join(', ') };
  }

  // 4. Update photo status to approved
  const { error: updateError } = await supabase
    .from('photos')
    .update({
      status: 'approved',
      approved_at: new Date().toISOString(),
      approved_by: user.id,
    })
    .eq('id', validation.data.photoId)
    .eq('status', 'pending'); // Only approve pending photos

  if (updateError) {
    console.error('Error approving photo:', updateError);
    return { success: false, error: 'Failed to approve photo' };
  }

  // 5. Revalidate paths
  revalidatePath('/admin/photos');
  revalidatePath('/admin');
  // Note: Can't revalidate specific cafe page without knowing cafe_id
  // The cafe detail page will pick up the change on next load

  return { success: true };
}

// REJECT PHOTO
// ============================================

const rejectPhotoSchema = z.object({
  photoId: z.string().uuid(),
  rejectionReason: z.string().min(5, 'Reason must be at least 5 characters'),
});

export async function rejectPhoto(input: z.infer<typeof rejectPhotoSchema>): Promise<{
  success: boolean;
  error?: string;
}> {
  // 1. Verify authentication
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();

  if (!user) {
    return { success: false, error: 'Authentication required' };
  }

  // 2. Verify admin role
  if (!(await verifyAdminRole(supabase, user.id))) {
    return { success: false, error: 'Unauthorized - admin role required' };
  }

  // 3. Validate input
  const validation = rejectPhotoSchema.safeParse(input);
  if (!validation.success) {
    return { success: false, error: validation.error.issues.map(i => i.message).join(', ') };
  }

  // 4. Update photo status to rejected
  const { error: updateError } = await supabase
    .from('photos')
    .update({
      status: 'rejected',
      rejection_reason: validation.data.rejectionReason,
    })
    .eq('id', validation.data.photoId)
    .eq('status', 'pending'); // Only reject pending photos

  if (updateError) {
    console.error('Error rejecting photo:', updateError);
    return { success: false, error: 'Failed to reject photo' };
  }

  // 5. Revalidate paths
  revalidatePath('/admin/photos');
  revalidatePath('/admin');

  return { success: true };
}

// GET PENDING PHOTOS (for admin table)
// ============================================

export async function getPendingPhotos(options?: {
  offset?: number;
  limit?: number;
}): Promise<{
  success: boolean;
  photos?: any[];
  total?: number;
  error?: string;
}> {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();

  if (!user) {
    return { success: false, error: 'Authentication required' };
  }

  if (!(await verifyAdminRole(supabase, user.id))) {
    return { success: false, error: 'Unauthorized' };
  }

  const offset = options?.offset ?? 0;
  const limit = Math.min(options?.limit ?? 20, 50);

  const { data: photos, error, count } = await supabase
    .from('photos')
    .select(`
      id,
      storage_path,
      status,
      created_at,
      user_id,
      cafe_id,
      cafe:cafes(id, name, slug)
    `, { count: 'exact' })
    .eq('status', 'pending')
    .order('created_at', { ascending: false })
    .range(offset, offset + limit - 1);

  if (error) {
    console.error('Error fetching pending photos:', error);
    return { success: false, error: 'Failed to fetch photos' };
  }

  return {
    success: true,
    photos: photos || [],
    total: count || 0,
  };
}

'use client';

import { useState, useTransition } from 'react';
import Image from 'next/image';
import { Button } from '@/components/ui/button';
import { Card } from '@/components/ui/card';
import { PhotoPreviewModal } from './photo-preview-modal';
import { RejectPhotoModal } from './reject-photo-modal';
import { approvePhoto } from '@/lib/actions/admin';
import { toast } from 'sonner';
import { Eye, Check, X } from 'lucide-react';

interface PendingPhoto {
  id: string;
  storage_path: string;
  created_at: string;
  cafe: { id: string; name: Record<string, string>; slug: string } | null;
}

interface PhotosTableProps {
  photos: PendingPhoto[];
  translations: Record<string, string>;
  storageUrl: string; // Base URL for Supabase Storage
}

export function PhotosTable({ photos, translations, storageUrl }: PhotosTableProps) {
  const [previewPhoto, setPreviewPhoto] = useState<PendingPhoto | null>(null);
  const [rejectPhoto, setRejectPhoto] = useState<PendingPhoto | null>(null);
  const [isPending, startTransition] = useTransition();

  const handleApprove = (photoId: string) => {
    startTransition(async () => {
      const result = await approvePhoto({ photoId });
      if (result.success) {
        toast.success(translations['admin.success.photoApproved']);
      } else {
        toast.error(result.error || translations['admin.error.generic']);
      }
    });
  };

  const getPhotoUrl = (storagePath: string, width = 200) => {
    // Use Supabase Storage transform for thumbnails
    return `${storageUrl}/photos/${storagePath}?width=${width}&height=${width}&resize=cover`;
  };

  return (
    <>
      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-4">
        {photos.map((photo) => (
          <Card key={photo.id} className="overflow-hidden">
            <div className="relative aspect-square">
              <Image
                src={getPhotoUrl(photo.storage_path)}
                alt="Pending photo"
                fill
                className="object-cover"
                sizes="(max-width: 768px) 50vw, (max-width: 1024px) 33vw, 25vw"
              />
            </div>
            <div className="p-2">
              <p className="text-sm font-medium truncate">
                {photo.cafe?.name?.en || photo.cafe?.name?.ko || 'Unknown cafe'}
              </p>
              <p className="text-xs text-muted-foreground">
                {new Date(photo.created_at).toLocaleDateString()}
              </p>
              <div className="flex gap-1 mt-2">
                <Button
                  size="sm"
                  variant="outline"
                  onClick={() => setPreviewPhoto(photo)}
                  title={translations['admin.photos.preview']}
                >
                  <Eye className="h-4 w-4" />
                </Button>
                <Button
                  size="sm"
                  onClick={() => handleApprove(photo.id)}
                  disabled={isPending}
                  title={translations['admin.photos.approve']}
                  className="bg-green-600 hover:bg-green-700"
                >
                  <Check className="h-4 w-4" />
                </Button>
                <Button
                  size="sm"
                  variant="destructive"
                  onClick={() => setRejectPhoto(photo)}
                  title={translations['admin.photos.reject']}
                >
                  <X className="h-4 w-4" />
                </Button>
              </div>
            </div>
          </Card>
        ))}
      </div>

      {/* Preview Modal */}
      {previewPhoto && (
        <PhotoPreviewModal
          photo={previewPhoto}
          isOpen={!!previewPhoto}
          onClose={() => setPreviewPhoto(null)}
          onApprove={() => handleApprove(previewPhoto.id)}
          onReject={() => {
            setPreviewPhoto(null);
            setRejectPhoto(previewPhoto);
          }}
          storageUrl={storageUrl}
          translations={translations}
        />
      )}

      {/* Reject Modal */}
      {rejectPhoto && (
        <RejectPhotoModal
          photoId={rejectPhoto.id}
          isOpen={!!rejectPhoto}
          onClose={() => setRejectPhoto(null)}
          translations={translations}
        />
      )}
    </>
  );
}
