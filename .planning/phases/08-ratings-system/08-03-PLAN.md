## Key Requirements

**RATE-05 (Update/Overwrite):**
- Upsert pattern: INSERT ON CONFLICT (user_id, cafe_id) DO UPDATE
- Track updated_at timestamp
- Return the updated rating

**RATE-04 (Zero exclusion from averages):**
- SQL: AVG(NULLIF(dimension, 0))
- Or calculate in application code excluding 0s
- Update cafes table overall_rating and total_ratings

**RATE-08 (No rate limit):**
- No rate limit checks unlike submissions
- Users can rate unlimited cafes
- Only one rating per cafe (per user) enforced by constraint

**Authentication:**
- All actions require authenticated user
- Verify user_id matches auth.uid()
- Return { success: false, error: string } on auth failure

**src/app/api/ratings/route.ts:**

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { submitRating, getCafeRatingsList } from '@/lib/actions/ratings';

/**
 * GET /api/ratings?cafeId=xxx
 * Get ratings for a specific cafe
 */
export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url);
    const cafeId = searchParams.get('cafeId');
    const limit = parseInt(searchParams.get('limit') || '20', 10);
    const offset = parseInt(searchParams.get('offset') || '0', 10);

    if (!cafeId) {
      return NextResponse.json(
        { error: 'cafeId query parameter required' },
        { status: 400 }
      );
    }

    const result = await getCafeRatingsList(cafeId, { limit, offset });

    if (!result.success) {
      return NextResponse.json(
        { error: result.error },
        { status: 500 }
      );
    }

    return NextResponse.json({
      ratings: result.ratings,
      total: result.total,
      limit,
      offset,
    });
  } catch (error) {
    console.error('Error fetching ratings:', error);
    return NextResponse.json(
      { error: 'Failed to fetch ratings' },
      { status: 500 }
    );
  }
}

/**
 * POST /api/ratings
 * Submit or update a rating
 */
export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const result = await submitRating(body);

    if (!result.success) {
      return NextResponse.json(
        { error: result.error },
        { status: result.error?.includes('Authentication') ? 401 : 400 }
      );
    }

    return NextResponse.json(result.rating, { status: 201 });
  } catch (error) {
    console.error('Error submitting rating:', error);
    return NextResponse.json(
      { error: 'Failed to submit rating' },
      { status: 500 }
    );
  }
}
```
