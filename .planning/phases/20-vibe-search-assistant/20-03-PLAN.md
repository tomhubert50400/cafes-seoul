2. Create a preset selection handler inside the component that maps preset filter values to URL params:
```typescript
const handlePresetSelect = useCallback((presetId: string) => {
  const preset = FILTER_PRESETS.find(p => p.id === presetId);
  if (!preset) return;

  // Build new URL params from preset filters
  const params = new URLSearchParams();

  // Map MapFilters keys to URL param names
  // The cafes list API uses these URL params:
  // hasWifi, hasOutlets, isPetFriendly, isLaptopFriendly, hasParking
  // minRating (overall), district, cafeType, sortBy, q
  // Note: The cafes list page API does NOT support per-dimension rating filters
  // (seatingMin, wifiMin, etc.) -- those are map-only via client-side filtering.
  // So for the list page, we only apply the boolean feature filters from the preset.

  if (preset.filters.hasWifi) params.set('hasWifi', 'true');
  if (preset.filters.hasPowerOutlets) params.set('hasOutlets', 'true');
  if (preset.filters.isPetFriendly) params.set('isPetFriendly', 'true');
  if (preset.filters.isLaptopFriendly) params.set('isLaptopFriendly', 'true');
  if (preset.filters.hasParking) params.set('hasParking', 'true');

  // Preserve current sort preference
  const currentSort = searchParams.get('sortBy');
  if (currentSort) params.set('sortBy', currentSort);

  router.push(`?${params.toString()}`);
}, [router, searchParams]);
```

3. Compute the matched preset from current URL params:
```typescript
const matchedPresetId = useMemo(() => {
  // Build a partial MapFilters from current URL params
  const currentFilters = {
    hasWifi: searchParams.get('hasWifi') === 'true',
    hasPowerOutlets: searchParams.get('hasOutlets') === 'true',
    isPetFriendly: searchParams.get('isPetFriendly') === 'true',
    isLaptopFriendly: searchParams.get('isLaptopFriendly') === 'true',
    hasParking: searchParams.get('hasParking') === 'true',
  };

  // Check each preset -- only compare boolean filters (list page limitation)
  for (const preset of FILTER_PRESETS) {
    const booleanKeys = ['hasWifi', 'hasPowerOutlets', 'isPetFriendly', 'isLaptopFriendly', 'hasParking'] as const;
    const presetBooleans: Record<string, boolean> = {};
    const currentBooleans: Record<string, boolean> = {};

    for (const key of booleanKeys) {
      presetBooleans[key] = preset.filters[key] === true;
      currentBooleans[key] = currentFilters[key] === true;
    }

    // Check if preset booleans match AND no extra booleans active
    const presetMatches = booleanKeys.every(key => presetBooleans[key] === currentBooleans[key]);

    // Also check no other non-preset URL params are active (district, cafeType, q)
    const hasExtraFilters = searchParams.get('district') || searchParams.get('cafeType') || searchParams.get('q');

    if (presetMatches && !hasExtraFilters) return preset.id;
  }

  return null;
}, [searchParams]);
```
