1. Create src/lib/supabase/auth-helpers.ts:
   ```typescript
   import { SupabaseClient } from '@supabase/supabase-js';

   export async function hasPasswordAuth(supabase: SupabaseClient): Promise<boolean> {
     const { data: { user } } = await supabase.auth.getUser();
     if (!user) return false;

     const { data: identities } = await supabase.auth.getUserIdentities();
     if (!identities?.identities) return false;

     // Check if user has email provider (not just OAuth)
     return identities.identities.some(
       identity => identity.provider === 'email'
     );
   }

   export async function getUserEmail(supabase: SupabaseClient): Promise<string | null> {
     const { data: { user } } = await supabase.auth.getUser();
     return user?.email ?? null;
   }
   ```
