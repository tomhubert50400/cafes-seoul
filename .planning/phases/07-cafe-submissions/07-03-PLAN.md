- Rate limit: 3 submissions per user per day

## Technical Context

**Existing patterns:**
- Server Actions in src/lib/actions/ (auth pattern)
- Supabase client creation in src/lib/supabase/server.ts
- RLS policies handle base authorization
- Rate limiting needs application-level enforcement

**Rate limiting strategy:**
1. Check submission_rate_limits table for user
2. If no record or reset_at passed: create new record with count=1
3. If count < 3: increment count
4. If count >= 3: reject with rate limit error
5. Reset at midnight KST (or 24h from first submission)

**Duplicate detection strategy:**
- Fuzzy matching on name similarity (trigram similarity)
- Address substring matching
- Return top 5 potential matches
- Called before final submission (user can proceed anyway)

1. **getRateLimit(supabase, userId):**
   ```typescript
   export async function getRateLimit(
     supabase: SupabaseClient,
     userId: string
   ): Promise<SubmissionRateLimit | null> {
     const { data, error } = await supabase
       .from('submission_rate_limits')
       .select('*')
       .eq('user_id', userId)
       .single();
     
     if (error || !data) return null;
     
     // Check if reset time has passed
     const resetAt = new Date(data.reset_at);
     if (resetAt < new Date()) {
       // Reset has passed, delete old record
       await supabase
         .from('submission_rate_limits')
         .delete()
         .eq('user_id', userId);
       return null;
     }
     
     return transformSubmissionRateLimit(data);
   }
   ```

2. **incrementRateLimit(supabase, userId):**
   ```typescript
   export async function incrementRateLimit(
     supabase: SupabaseClient,
     userId: string
   ): Promise<{ success: boolean; rateLimit?: SubmissionRateLimit; error?: string }> {
     const now = new Date();
     const tomorrow = new Date(now);
     tomorrow.setDate(tomorrow.getDate() + 1);
     tomorrow.setHours(0, 0, 0, 0); // Midnight KST
     
     // Try to get existing
     const existing = await getRateLimit(supabase, userId);
     
     if (!existing) {
       // First submission today
       const { data, error } = await supabase
         .from('submission_rate_limits')
         .insert({
           user_id: userId,
           submission_count: 1,
           reset_at: tomorrow.toISOString(),
           last_submission_at: now.toISOString(),
         })
         .select()
         .single();
       
       if (error) return { success: false, error: error.message };
       return { success: true, rateLimit: transformSubmissionRateLimit(data) };
     }
     
     if (existing.submissionCount >= 3) {
       return { 
         success: false, 
         error: 'Daily submission limit reached',
         rateLimit: existing
       };
     }
     
     // Increment
     const { data, error } = await supabase
       .from('submission_rate_limits')
       .update({
         submission_count: existing.submissionCount + 1,
         last_submission_at: now.toISOString(),
       })
       .eq('user_id', userId)
       .select()
       .single();
     
     if (error) return { success: false, error: error.message };
     return { success: true, rateLimit: transformSubmissionRateLimit(data) };
   }
   ```

**NOTES:**
- Reset is at midnight KST (Korean time)
- Limit is strictly 3 per day
- Returns full rate limit info for UI display
- Thread-safe via database row locking (single row per user)

   ```typescript
   export async function submitCafe(
     data: SubmissionFormData
   ): Promise<{ success: boolean; submission?: CafeSubmission; error?: string; rateLimit?: SubmissionRateLimit }> {
     // 1. Verify user is authenticated
     const supabase = await createClient();
     const { data: { user } } = await supabase.auth.getUser();
     if (!user) return { success: false, error: 'Authentication required' };
     
     // 2. Check rate limit
     const rateCheck = await incrementRateLimit(supabase, user.id);
     if (!rateCheck.success) {
       return { success: false, error: rateCheck.error, rateLimit: rateCheck.rateLimit };
     }
     
     // 3. Validate input
     const validation = submissionSchema.safeParse(data);
     if (!validation.success) {
       return { success: false, error: 'Validation failed' };
     }
     
     // 4. Create submission
     const result = await createSubmission(supabase, user.id, validation.data);
     if ('error' in result) {
       return { success: false, error: result.error };
     }
     
     // 5. Return success with rate limit info
     return { success: true, submission: result, rateLimit: rateCheck.rateLimit };
   }
   ```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/server';
import { findDuplicateCafes } from '@/lib/supabase/submissions';
import type { TranslatedText } from '@/types/cafe';

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const { name, address } = body as { name: TranslatedText; address: TranslatedText };
    
    // Validation
    if (!name || !address) {
      return NextResponse.json(
        { error: 'Name and address are required' },
        { status: 400 }
      );
    }
    
    // Must have at least one language
    const hasName = Object.values(name).some(v => v && v.trim().length > 0);
    if (!hasName) {
      return NextResponse.json(
        { error: 'At least one name translation is required' },
        { status: 400 }
      );
    }
    
    const supabase = await createClient();
    const duplicates = await findDuplicateCafes(supabase, name, address);
    
    return NextResponse.json({ duplicates });
  } catch (error) {
    console.error('Error checking duplicates:', error);
    return NextResponse.json(
      { error: 'Failed to check for duplicates' },
      { status: 500 }
    );
  }
}
```
